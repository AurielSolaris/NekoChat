cmake_minimum_required(VERSION 3.22)
project(nekochat CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(SHARED_DIR ${CMAKE_CURRENT_SOURCE_DIR}/shared/cpp)
set(GPT2_DIR ${CMAKE_CURRENT_SOURCE_DIR}/gpt2/cpp)
set(QWEN3_DIR ${CMAKE_CURRENT_SOURCE_DIR}/qwen3/cpp)
set(GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated)

# ---------------------------------------------------------------------------
# GPU kernels: one GLSL source per kernel, used twice:
#   Vulkan  -> compiled to SPIR-V with the NDK's glslc at configure time
#   GLES    -> embedded as source text and compiled by the driver at runtime
# Order must match neko::gpu::Kernel.
set(NEKO_KERNELS embed layernorm matmul kv_store attention rope silu_mul matmul_q8 matmul_q4)
set(SHADER_DIR ${SHARED_DIR}/gpu/shaders)

if(CMAKE_HOST_WIN32)
    set(NDK_HOST windows-x86_64)
    set(GLSLC_NAME glslc.exe)
elseif(CMAKE_HOST_APPLE)
    set(NDK_HOST darwin-x86_64)
    set(GLSLC_NAME glslc)
else()
    set(NDK_HOST linux-x86_64)
    set(GLSLC_NAME glslc)
endif()
find_program(GLSLC ${GLSLC_NAME} HINTS ${ANDROID_NDK}/shader-tools/${NDK_HOST} ${CMAKE_ANDROID_NDK}/shader-tools/${NDK_HOST}
             NO_DEFAULT_PATH)
if(NOT GLSLC)
    message(FATAL_ERROR "glslc not found in the NDK (shader-tools/${NDK_HOST})")
endif()

file(MAKE_DIRECTORY ${GEN_DIR})
file(READ ${SHADER_DIR}/common.glsl SHADER_COMMON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${SHADER_DIR}/common.glsl)
set(GEN "// Generated by architecture/CMakeLists.txt from shared/cpp/gpu/shaders. Do not edit.\n#pragma once\n#include <cstddef>\n\n")
set(GEN_GLSL "")
set(GEN_SPV "")
set(GEN_SIZE "")
set(GEN_NAMES "")
foreach(K ${NEKO_KERNELS})
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${SHADER_DIR}/${K}.comp)
    file(READ ${SHADER_DIR}/${K}.comp BODY)
    file(WRITE ${GEN_DIR}/${K}.vk.comp "#version 450\n#define NEKO_VULKAN 1\n${SHADER_COMMON}\n${BODY}")
    execute_process(
        COMMAND ${GLSLC} -O --target-env=vulkan1.0 -fshader-stage=compute ${GEN_DIR}/${K}.vk.comp -o ${GEN_DIR}/${K}.spv
        RESULT_VARIABLE RC ERROR_VARIABLE ERR)
    if(NOT RC EQUAL 0)
        message(FATAL_ERROR "glslc failed for ${K}.comp:\n${ERR}")
    endif()
    file(READ ${GEN_DIR}/${K}.spv HEX HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," HEX "${HEX}")
    string(APPEND GEN "alignas(4) static const unsigned char kSpv_${K}[] = {${HEX}};\n")
    string(APPEND GEN "static const char kGlsl_${K}[] = R\"NEKO(${SHADER_COMMON}\n${BODY})NEKO\";\n\n")
    string(APPEND GEN_GLSL "kGlsl_${K}, ")
    string(APPEND GEN_SPV "kSpv_${K}, ")
    string(APPEND GEN_SIZE "sizeof(kSpv_${K}), ")
    string(APPEND GEN_NAMES "\"${K}\", ")
endforeach()
string(APPEND GEN "static const char* const kShaderGlsl[] = {${GEN_GLSL}};\n")
string(APPEND GEN "static const unsigned char* const kShaderSpv[] = {${GEN_SPV}};\n")
string(APPEND GEN "static const size_t kShaderSpvSize[] = {${GEN_SIZE}};\n")
string(APPEND GEN "static const char* const kShaderNames[] = {${GEN_NAMES}};\n")
file(WRITE ${GEN_DIR}/shaders_gen.h.tmp "${GEN}")
configure_file(${GEN_DIR}/shaders_gen.h.tmp ${GEN_DIR}/shaders_gen.h COPYONLY)

# ---------------------------------------------------------------------------
add_library(neko_core STATIC
    ${SHARED_DIR}/json.cpp
    ${SHARED_DIR}/tensor_store.cpp
    ${SHARED_DIR}/quant.cpp
    ${SHARED_DIR}/tokenizer.cpp
    ${SHARED_DIR}/thread_pool.cpp
    ${SHARED_DIR}/cpu_ops.cpp
    ${SHARED_DIR}/sampler.cpp
    ${SHARED_DIR}/engine.cpp
    ${SHARED_DIR}/model.cpp
    ${SHARED_DIR}/gpu/compute.cpp
    ${SHARED_DIR}/gpu/vk_compute.cpp
    ${SHARED_DIR}/gpu/gl_compute.cpp
    ${GPT2_DIR}/gpt2_model.cpp
    ${QWEN3_DIR}/qwen3_model.cpp
)
target_include_directories(neko_core PUBLIC ${SHARED_DIR} ${GPT2_DIR} ${QWEN3_DIR} ${GEN_DIR})
# Inference must be optimised even in debug APKs.
target_compile_options(neko_core PRIVATE -O3 -Wall -Wno-unused-function)
target_link_libraries(neko_core PUBLIC log EGL GLESv3 dl)

add_library(nekochat SHARED ${SHARED_DIR}/jni_bridge.cpp)
target_link_libraries(nekochat PRIVATE neko_core)
target_link_options(nekochat PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384)

option(NEKO_BUILD_CLI "Build the on-device validation tool" OFF)
if(NEKO_BUILD_CLI)
    add_executable(nekochat_cli ${SHARED_DIR}/tools/cli_main.cpp)
    target_link_libraries(nekochat_cli PRIVATE neko_core)
endif()
